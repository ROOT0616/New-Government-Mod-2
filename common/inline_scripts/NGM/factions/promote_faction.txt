actions = {
	# First promoted faction of this ethic also promotes the empire-wide ethic attraction
	promote_faction = {
		title = "PROMOTE_FACTION"
		description = "PROMOTE_FACTION_DESC"

		potential = {
			NOR = {
				has_pop_faction_flag = promoted_faction
				has_modifier = suppressed_faction
			}
			owner = { NOT = { has_modifier = promoted_$ETHIC$ } }
		}

		effect = {
			set_pop_faction_flag = promoted_faction
			owner = {
				set_country_flag = has_promoted_faction
				add_modifier = { modifier = promoted_$ETHIC$ days = -1 }
			}
		}

		ai_weight = {
			base = 0
		}
	}
	# Another faction of this ethic is already promoted
	promote_faction_shared = {
		title = "PROMOTE_FACTION"
		description = "PROMOTE_FACTION_SHARED_DESC"

		potential = {
			NOR = {
				has_pop_faction_flag = promoted_faction
				has_modifier = suppressed_faction
			}
			owner = { has_modifier = promoted_$ETHIC$ }
		}

		effect = {
			set_pop_faction_flag = promoted_faction
			owner = {
				set_country_flag = has_promoted_faction
			}
		}

		ai_weight = {
			base = 0
		}
	}

	cancel_promote_faction = {
		title = "CANCEL_PROMOTE_FACTION"
		description = "CANCEL_PROMOTE_FACTION_DESC"

		potential = {
			has_pop_faction_flag = promoted_faction
		}

		effect = {
			remove_pop_faction_flag = promoted_faction
			owner = {
				if = {
					limit = { # Only remove the modifier if no other faction with the same ethic is promoted
						NOT = {
							any_pop_faction = {
								is_pop_faction_type = $FACTION$
								has_pop_faction_flag = promoted_faction
							}
						}
					}
					remove_modifier = promoted_$ETHIC$
				}
			}
		}

		ai_weight = {
			base = 0
		}
	}
}
