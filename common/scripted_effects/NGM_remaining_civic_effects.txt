ngm_apprentice_leader_effect = {
	if = {
		limit = {
			root = {
				has_valid_civic = civic_ngm_apprenticeship
			}
			OR = {
				leader_class = official
				leader_class = commander
				leader_class = scientist
			}
			NOT = {
				has_leader_flag = ngm_apprentice_initialized
			}
		}
		if = {
			limit = {
				has_base_skill < 2
			}
			set_skill = 2
		}
		add_trait = {
			trait = random_common
			show_message = no
		}
		set_leader_flag = ngm_apprentice_initialized
	}
}
ngm_insanity_leader_effect = {
	if = {
		limit = {
			root = {
				has_valid_civic = civic_ngm_fundamental_insanity
			}
		}
		if = {
			limit = {
				leader_class = scientist
				NOR = {
					has_trait = leader_trait_maniacal
					has_trait = leader_trait_maniacal_2
					has_trait = leader_trait_maniacal_3
				}
			}
			add_trait = {
				trait = leader_trait_maniacal
				show_message = no
			}
		}
		if = {
			limit = {
				leader_class = commander
				NOR = {
					has_trait = leader_trait_trickster
				}
			}
			add_trait = {
				trait = leader_trait_trickster
				show_message = no
			}
		}
	}
}

ngm_youth_species_effect = {
	if = {
		limit = {
			has_valid_civic = civic_ngm_youth_politics
			species = {
				NOT = {
					has_trait = trait_ngm_youth_politics
				}
			}
		}
		species = {
			save_event_target_as = ngm_old_political_species
			modify_species = {
				species = this
				base = auto
				add_trait = trait_ngm_youth_politics
				remove_trait = trait_ngm_elder_politics
				change_scoped_species = no
				effect = {
					save_event_target_as = ngm_new_political_species
				}
			}
		}
		every_owned_pop_group = {
			limit = {
				exists = species
				is_exact_same_species = event_target:ngm_old_political_species
			}
			change_species = event_target:ngm_new_political_species
		}
		every_owned_leader = {
			limit = {
				exists = species
				is_exact_same_species = event_target:ngm_old_political_species
			}
			change_species = event_target:ngm_new_political_species
		}
		every_pool_leader = {
			limit = {
				exists = species
				is_exact_same_species = event_target:ngm_old_political_species
			}
			change_species = event_target:ngm_new_political_species
		}
		every_envoy = {
			limit = {
				exists = species
				is_exact_same_species = event_target:ngm_old_political_species
			}
			change_species = event_target:ngm_new_political_species
		}
		every_owned_army = {
			limit = {
				exists = species
				is_exact_same_species = event_target:ngm_old_political_species
			}
			change_species = event_target:ngm_new_political_species
		}
		every_controlled_ship = {
			limit = {
				is_ship_class = shipclass_colonizer
				is_exact_same_species = event_target:ngm_old_political_species
			}
			change_species = event_target:ngm_new_political_species
		}
		change_dominant_species = {
			species = event_target:ngm_new_political_species
		}
	}
}
