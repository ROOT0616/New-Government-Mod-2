# Country scope; PREVPREV in defender scope refers to this country.
ngm_sync_adversity_effect = {
 if = { limit = { has_tradition = tr_ngm_adversity_finish is_at_war = yes
  any_war = { any_defender = { is_same_value = prevprev } }
 }
  if = { limit = { NOT = { has_modifier = tr_ngm_adversity_finish_modifier } }
   add_modifier = { modifier = tr_ngm_adversity_finish_modifier days = -1 }
  }
 } else = {
  if = { limit = { has_modifier = tr_ngm_adversity_finish_modifier } remove_modifier = tr_ngm_adversity_finish_modifier }
 }
}
