            set_timed_country_flag = { flag = ngm_prosperity_interval days = 1825 }
        }
        else_if = { limit = { NOT = { has_country_flag = ngm_prosperity_interval } NOT = { has_country_flag = ngm_prosperity_tier_10 } }
            if = { limit = { has_country_flag = ngm_prosperity_tier_9 }
        remove_country_flag = ngm_prosperity_tier_9 set_country_flag = ngm_prosperity_tier_10
    }
else_if = { limit = { has_country_flag = ngm_prosperity_tier_8 }
        remove_country_flag = ngm_prosperity_tier_8 set_country_flag = ngm_prosperity_tier_9
    }
