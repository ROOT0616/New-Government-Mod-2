ngm_dissection_species_can_be_enslaved = {
	exists = root
	NAND = {
		exists = this
		OR = {
			root = {
				is_wilderness_empire = yes
			}
			AND = {
				root = {
					is_individual_machine = yes
				}
				OR = {
					has_trait = trait_machine_unit
					has_trait = trait_mechanical
				}
			}
			has_trait = trait_machine_unit
			has_trait = trait_mechanical
		}
	}
	is_disconnected_drone_pop = no
	NAND = {
		root = {
			has_origin = origin_forever_cruise
			NOT = {
				has_country_flag = forever_cruise_over
			}
		}
		OR = {
			is_founding_cruise_crew = yes
			is_founding_cruise_passenger = yes
		}
	}
	NOT = {
		root = {
			has_origin = origin_nomadic_purger
		}
	}
	hidden_trigger = {
		exists = root
	}
	custom_tooltip = {
		fail_text = presapients_allowed_citizenship
		OR = {
			is_sapient = yes
			root = {
				NOT = {
					has_policy_flag = pre_sapients_protect
					has_policy_flag = pre_sapients_allow
				}
			}
		}
	}
	custom_tooltip = {
		fail_text = presapients_purged_citizenship
		OR = {
			is_sapient = yes
			root = {
				NOT = {
					has_policy_flag = pre_sapients_purge
				}
			}
		}
	}
	custom_tooltip = {
		fail_text = HIVE_MIND_MAIN_SPECIES_ALWAYS_CITIZEN
		NAND = {
			root = {
				is_hive_empire = yes
			}
			has_trait = trait_hive_mind
		}
	}
	custom_tooltip = {
		fail_text = WILDERNESS_DEAD_WORLD
		if = {
			limit = {
				root = {
					is_wilderness_empire = no
				}
			}
			NOT = {
				has_trait = trait_wilderness
			}
		}
	}
	custom_tooltip = {
		fail_text = same_species_disallowed
		NOT = {
			is_same_species = root
		}
	}
	custom_tooltip = {
		fail_text = lonely_hive_mind_pops_kill_themselves_citizenship
		is_lonely_hive_mind_pop = no
	}
	custom_tooltip = {
		fail_text = SERVITORS_CANNOT_ENSLAVE
		root = {
			NOT = {
				has_valid_civic = civic_machine_servitor
			}
		}
	}
	custom_tooltip = {
		fail_text = ASSIMILATORS_CANNOT_ENSLAVE
		root = {
			NOT = {
				has_valid_civic = civic_machine_assimilator
			}
		}
	}
	custom_tooltip = {
		fail_text = TERMINATORS_CANNOT_ENSLAVE
		root = {
			NOT = {
				has_valid_civic = civic_machine_terminator
			}
		}
	}
	custom_tooltip = {
		fail_text = DEVOURERS_CANNOT_ENSLAVE
		root = {
			NOT = {
				has_valid_civic = civic_hive_devouring_swarm
			}
		}
	}
	custom_tooltip = {
		fail_text = empire_xenophile
		root = {
			is_xenophile = no
		}
	}
	custom_tooltip = {
		fail_text = FANATIC_PURIFIERS_MUST_PURIFY
		OR = {
			NOT = {
				root = {
					has_valid_civic = civic_fanatic_purifiers
				}
			}
			is_same_species = root
		}
	}
	custom_tooltip = {
		fail_text = does_not_embrace_species_slavery
		root = {
			allows_slavery = yes
			OR = {
				is_xenophobe = yes
				is_authoritarian = yes
				has_ethic = ethic_gestalt_consciousness
			}
		}
	}
	custom_tooltip = {
		fail_text = yes_service_yes_citizenship
		if = {
			limit = {
				root = {
					has_valid_civic = civic_citizen_service
				}
			}
			NOT = {
				has_military_service_type = {
					type = military_service_full
					country = root
				}
			}
		}
	}
	custom_tooltip = {
		fail_text = selective_kinship_must_citizen
		if = {
			limit = {
				root = {
					has_valid_civic = civic_selective_kinship
				}
			}
			NOT = {
				is_same_species_class = root
			}
		}
	}
}
